/**
 * Keeps the DashCache copies warm. Every 5 minutes. Does the work only from 08:00 to 20:00 Asia/Manila.
 * No new OAuth scope. Requires DashCache.gs.
 * Run installDashWarm once. Run removeDashWarm to stop it.
 *
 * dashWarmBuild_ is the only function to edit once Dashboard.gs and DashboardDrill.gs are in hand.
 * Use the same calls Auth.gs already wraps. Pass `user` through. user.role is admin, production_manager, or accounting.
 */
function installDashWarm() {
  var gone = dashWarmClear_();
  ScriptApp.newTrigger('dashWarm_').timeBased().everyMinutes(5).create();
  return { ok: true, removed: gone, every: '5 minutes', hours: '08:00-20:00 Asia/Manila' };
}

function removeDashWarm() {
  return { ok: true, removed: dashWarmClear_() };
}

function dashWarmClear_() {
  var gone = 0;
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'dashWarm_') { ScriptApp.deleteTrigger(t); gone++; }
  });
  return gone;
}

function dashWarmOpen_() {
  var h = Number(Utilities.formatDate(new Date(), 'Asia/Manila', 'H'));
  return h >= 8 && h < 20;
}

function dashWarm_() {
  if (!dashWarmOpen_()) return { ok: true, skipped: 'closed' };
  if (typeof dashCacheRefresh_ !== 'function') return { ok: false, error: 'Paste DashCache.gs first' };
  var kept = dashCacheRefresh_();
  var built = dashWarmFill_();
  Logger.log(JSON.stringify({ ok: true, kept: kept, built: built }));
  return { ok: true, kept: kept, built: built };
}

function dashWarmFill_() {
  if (typeof dashCacheHas_ !== 'function' || typeof dashCachePut_ !== 'function') return [];
  var roles = ['admin', 'production_manager', 'accounting'];
  var jobs = [], r, built = [], i, job, user, fresh;
  for (r = 0; r < roles.length; r++) {
    jobs.push({ role: roles[r], action: 'dashboard', req: { action: 'dashboard' } });
    jobs.push({ role: roles[r], action: 'dashDrill', req: { action: 'dashDrill', what: 'jobs' } });
  }
  for (i = 0; i < jobs.length; i++) {
    job = jobs[i];
    if (typeof dashCacheSpec_ !== 'function') return built;
    job.spec = dashCacheSpec_(job.action, job.req);
    if (!job.spec || dashCacheHas_(job.role, job.action, job.req)) continue;
    user = { username: 'warm', role: job.role, display_name: 'Warm' };
    try { fresh = dashWarmBuild_(user, job.action, job.req); } catch (e) { fresh = null; }
    if (!fresh || !fresh.ok) continue;
    if (dashCachePut_(job.role, job.spec, fresh)) built.push(job.role + ' ' + (job.action === 'dashboard' ? 'dashboard' : 'jobs'));
  }
  return built;
}
