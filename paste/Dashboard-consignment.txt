/** Pieces still out on the one Consignment tab. Does not read Consignees, Consignments, ConsignmentItems, or Items. */
function dashConsignment_() {
  var today = new Date(Date.now() + 8 * 36e5).toISOString().slice(0, 10);
  var limit = 30, exec = false;
  try { exec = isExec_(); } catch (e0) { exec = false; }
  try {
    if (ss_().getSheetByName('Settings')) jobRows_('Settings').forEach(function (r) {
      if (String(r.key) !== 'consign_overdue_days') return;
      var v = parseInt(r.value, 10);
      if (v > 0 && v < 3650) limit = v;
    });
  } catch (e1) {}
  if (!ss_().getSheetByName('Consignment')) {
    return { missing: ['Consignment'], overdue_days: limit, consignments: [], counts: { consigned_out: 0, overdue_consignments: 0, consigned_jos: 0 } };
  }
  function day(v) {
    var s = (typeof jobDay_ === 'function') ? jobDay_(v) : String(v || '');
    return /^\d{4}-\d{2}-\d{2}/.test(s) ? s.slice(0, 10) : '';
  }
  function span(from, to) {
    if (!from || !to) return null;
    var a = Date.parse(from + 'T00:00:00Z'), b = Date.parse(to + 'T00:00:00Z');
    if (isNaN(a) || isNaN(b)) return null;
    return Math.round((b - a) / 86400000);
  }
  function open(st) { st = String(st || '').trim().toLowerCase(); return st === '' || st === 'consigned' || st === 'open' || st === 'out'; }
  function num(v) {
    if (v == null || String(v).trim() === '') return null;
    var n = Number(String(v).replace(/[₱,\s]/g, '').replace(/php/ig, ''));
    return isFinite(n) ? Math.round(n * 100) / 100 : null;
  }
  var list = [], onSheet = {};
  jobRows_('Consignment').forEach(function (r) {
    if (!open(r.status)) return;
    var dout = day(r.date_out), due = day(r.due_back || r.due_date), jo = (typeof jobKey_ === 'function' ? jobKey_(r.jo) : '') || String(r.jo || '').trim();
    var days = span(dout, today);
    var late = due ? today > due : (days != null && days > limit);
    var row = {
      description: String(r.description || ''), item_id: String(r.item_id || ''), jo: jo,
      consignee: String(r.consignee_name || r.consignee || ''), consignment_id: String(r.consignment_id || ''),
      date_out: dout, days_out: days, overdue: late
    };
    var agreed = num(r.agreed_value != null && String(r.agreed_value).trim() !== '' ? r.agreed_value : r.value);
    if (exec && agreed != null) row.value = agreed;
    if (jo) onSheet[jo] = true;
    list.push(row);
  });
  list.sort(function (a, b) { return String(a.date_out).localeCompare(String(b.date_out)) || String(a.consignment_id).localeCompare(String(b.consignment_id)); });
  var overdue = 0;
  list.forEach(function (r) { if (r.overdue) overdue++; });
  var loose = 0;
  if (ss_().getSheetByName('JobOrders')) jobRows_('JobOrders').forEach(function (j) {
    var k = (typeof jobKey_ === 'function') ? jobKey_(j.jo) : String(j.jo || '');
    if (!k || onSheet[k]) return;
    if (String(j.location || '').trim().toLowerCase() !== 'consigned') return;
    if (String(j.status || '').trim().toLowerCase() === 'cancelled') return;
    loose++;
  });
  return { missing: [], overdue_days: limit, consignments: list, counts: { consigned_out: list.length, overdue_consignments: overdue, consigned_jos: loose } };
}
