function dashConsign_(today, limit, exec, liveJobs, missing) {
  function joOf(v) {
    if (typeof jobKey_ === 'function') { try { var k = jobKey_(v); if (k) return String(k); } catch (e0) {} }
    return String(v || '').trim();
  }
  function num(v) {
    if (v == null || String(v).trim() === '') return null;
    var n = Number(String(v).replace(/[₱,\s]/g, '').replace(/php/ig, ''));
    return isFinite(n) ? Math.round(n * 100) / 100 : null;
  }
  function open(st) {
    st = String(st || '').trim().toLowerCase();
    return st === '' || st === 'consigned' || st === 'open' || st === 'out';
  }
  var rows = dashRead_('Consignment', missing);
  var list = [], onSheet = {}, owed = 0, i, r, st, jo, agreed, paid, dout, due, days, late, row, value;
  for (i = 0; i < rows.length; i++) {
    r = rows[i];
    st = String(r.status || '').trim().toLowerCase();
    jo = joOf(r.jo);
    if (st === 'sold' && exec) {
      agreed = num(r.agreed_value != null && String(r.agreed_value).trim() !== '' ? r.agreed_value : r.value);
      if (agreed != null) {
        paid = num(r.paid_amount);
        if (paid == null) paid = 0;
        owed = Math.round((owed + Math.max(0, agreed - paid)) * 100) / 100;
      }
    }
    if (!open(st)) continue;
    dout = dashDay_(r.date_out);
    due = dashDay_(r.due_back || r.due_date);
    days = dashDaysBetween_(dout, today);
    late = due ? today > due : (days != null && days > limit);
    row = {
      description: String(r.description || ''), item_id: String(r.item_id || ''), jo: jo,
      consignee: String(r.consignee_name || r.consignee || ''), consignment_id: String(r.consignment_id || ''),
      date_out: dout, days_out: days, overdue: !!late
    };
    value = num(r.agreed_value != null && String(r.agreed_value).trim() !== '' ? r.agreed_value : r.value);
    if (exec && value != null) row.value = value;
    if (jo) onSheet[jo] = 1;
    list.push(row);
  }
  list.sort(function (a, b) { return String(a.date_out).localeCompare(String(b.date_out)) || String(a.consignment_id).localeCompare(String(b.consignment_id)); });
  var overdueN = 0;
  list.forEach(function (item) { if (item.overdue) overdueN++; });
  var loose = 0;
  (liveJobs || []).forEach(function (j) {
    var k = joOf(j.jo);
    if (!k || onSheet[k]) return;
    if (String(j.location || '').trim().toLowerCase() !== 'consigned') return;
    loose++;
  });
  var counts = { consigned_out: list.length, overdue_consignments: overdueN, consigned_jos: loose };
  if (exec) counts.owed_by_consignees = owed;
  return { consignments: list, counts: counts };
}
