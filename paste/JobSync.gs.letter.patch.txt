function jobKey_(s) {
  var raw = String(s == null ? '' : s).trim();
  var t = raw.toUpperCase().replace(/\s+/g, '');
  var m = t.match(/^(\d{2})-(\d{3,4})([A-Z])?$/);
  if (m) return m[1] + '-' + m[2] + (m[3] || '');
  return raw;
}

function jobIsRevision_(j) {
  if (!j) return false;
  if (String(j.type || '').toLowerCase() === 'revision') return true;
  return !!String(j.parent_jo || '').trim();
}

function jobMergeVersions_(rows) {
  var out = [], grouped = {}, i, j, key, m, base, lettered, bucket, best, n, bKey, merged, hit;
  rows = rows || [];
  for (i = 0; i < rows.length; i++) {
    j = rows[i] || {};
    key = jobKey_(j.jo);
    m = String(key).match(/^(\d{2}-\d{3,4})([A-Z])?$/);
    base = m ? m[1] : '';
    lettered = !!(m && m[2]);
    if (!(lettered && jobIsRevision_(j))) { out.push(j); continue; }
    if (String(j.parent_jo || '').trim()) base = jobKey_(j.parent_jo);
    if (!base) { out.push(j); continue; }
    if (!grouped[base]) grouped[base] = [];
    grouped[base].push(j);
  }
  Object.keys(grouped).forEach(function (baseKey) {
    bucket = grouped[baseKey];
    best = bucket[0];
    for (n = 1; n < bucket.length; n++) {
      bKey = jobKey_(bucket[n].jo);
      if (bKey > jobKey_(best.jo)) best = bucket[n];
    }
    merged = {};
    Object.keys(best).forEach(function (k) { merged[k] = best[k]; });
    merged.jo = baseKey;
    merged.version = jobKey_(best.jo).slice(baseKey.length);
    hit = false;
    for (n = 0; n < out.length; n++) {
      if (jobKey_(out[n] && out[n].jo) === baseKey && !jobIsRevision_(out[n])) { out[n] = merged; hit = true; break; }
    }
    if (!hit) out.push(merged);
  });
  return out;
}

function jobRowFor_(rows, jo) {
  var want = jobKey_(jo), i;
  if (!want) return null;
  for (i = 0; i < (rows || []).length; i++) {
    if (rows[i] && jobKey_(rows[i].jo) === want) return rows[i];
  }
  return null;
}
