function scanZones_() {
  var raw = PropertiesService.getScriptProperties().getProperty('SCAN_ZONES');
  if (!raw) return [];
  try { return cleanScanZones_(JSON.parse(raw)) || []; } catch (e) { return []; }
}
function saveScanZones_(req) {
  var zones = cleanScanZones_(req.zones);
  if (!zones) return { ok: false, error: 'Bad zones' };
  PropertiesService.getScriptProperties().setProperty('SCAN_ZONES', JSON.stringify(zones));
  return { ok: true, saved: true, zones: zones };
}
function cleanScanZones_(raw) {
  if (Object.prototype.toString.call(raw) !== '[object Array]') return null;
  var known = { jo: 'JO number', due: 'Date due', given: 'Date given', client: 'Client', phone: 'Phone', item: 'Item', metal: 'Metal', karat: 'Karat', size: 'Size', stone: 'Stone', weight: 'Weight', price: 'Price', notes: 'Notes' };
  var out = [], seen = {}, used = {}, i, z, id, label, x, y, w, h, n, xm;
  for (i = 0; i < raw.length && out.length < 20; i++) {
    z = raw[i] || {};
    id = String(z.id || '');
    label = String(z.label || '').replace(/[\r\n]/g, ' ').replace(/\s+/g, ' ').trim();
    if (label.length > 24) label = label.slice(0, 24).trim();
    if (id === 'date') { id = 'due'; label = 'Date due'; }
    else if (/^other([2-4])?$/.test(id)) { if (!label) label = id === 'other' ? 'Other' : 'Other ' + id.slice(5); id = 'x'; }
    else if (known[id]) label = known[id];
    else if (!/^x(\d+)?$/.test(id)) continue;
    if (!label) label = 'Other';
    x = Number(z.x); y = Number(z.y); w = Number(z.w); h = Number(z.h);
    if (!(x >= 0 && y >= 0 && w > 0 && h > 0)) continue;
    if (x > 1 || y > 1 || w > 1 || h > 1) continue;
    if (x + w > 1) w = 1 - x;
    if (y + h > 1) h = 1 - y;
    if (w < 0.02 || h < 0.02) continue;
    xm = /^x(\d+)$/.exec(id);
    if (id === 'x' || xm) {
      n = xm ? Number(xm[1]) : 0;
      if (n < 1 || n > 6 || used[n]) { n = 0; for (var k = 1; k <= 6; k++) if (!used[k]) { n = k; break; } }
      if (!n) continue;
      id = 'x' + n; used[n] = true;
    } else {
      if (seen[id]) continue;
      seen[id] = true;
    }
    out.push({ id: id, label: label, x: Math.round(x * 10000) / 10000, y: Math.round(y * 10000) / 10000, w: Math.round(w * 10000) / 10000, h: Math.round(h * 10000) / 10000 });
  }
  return out;
}
function ocrCrops_(req, filename) {
  if (!req.ocrBase64) return null;
  var joText = ocr_(Utilities.newBlob(Utilities.base64Decode(String(req.ocrBase64)), 'image/jpeg', 'jo-' + filename), 'jo-' + filename);
  var ocrText = joText, ocrError = '', fields = req.ocrFields, i, ft, id, label, one;
  if (Object.prototype.toString.call(fields) !== '[object Array]') fields = [];
  for (i = 0; i < fields.length && i < 16; i++) {
    if (!fields[i] || !fields[i].image) continue;
    id = String(fields[i].id || 'box').replace(/[^a-z0-9]/gi, '').slice(0, 16).toLowerCase() || 'box';
    label = String(fields[i].label || id).replace(/[\r\n]/g, ' ').replace(/\s+/g, ' ').trim().slice(0, 24) || id;
    try {
      ft = ocr_(Utilities.newBlob(Utilities.base64Decode(String(fields[i].image)), 'image/jpeg', id + '-' + filename), id + '-' + filename);
      one = String(ft || '').replace(/[\r\n]+/g, ' ').replace(/\s+/g, ' ').trim().slice(0, 80);
      if (one) ocrText += '\nFIELD ' + id + ': ' + one + '\n' + label + ':\n' + ft;
    } catch (e2) { if (!ocrError) ocrError = String(e2); }
  }
  return { joText: joText, ocrText: ocrText, ocrError: ocrError };
}
function readJo_(req) {
  if (!req.ocrBase64) return { ok: false, error: 'Missing image' };
  if (String(req.ocrBase64).length > 2000000) return { ok: false, error: 'Too large' };
  var text = '', ocrError = '';
  try { text = ocr_(Utilities.newBlob(Utilities.base64Decode(String(req.ocrBase64)), 'image/jpeg', 'jo-read.jpg'), 'jo-read.jpg'); }
  catch (err) { ocrError = String(err); }
  var found = findJOs_(text);
  var compact = String(text || '').replace(/\s+/g, ' ').trim();
  var low = !(found.length === 1 && compact.length > 0 && compact.length <= 24);
  return { ok: true, jo: found.length ? found[0] : '', found: found, low: low, text: compact.slice(0, 80), ocrError: ocrError };
}
