function scanZones_() {
  var raw = PropertiesService.getScriptProperties().getProperty('SCAN_ZONES');
  if (!raw) return [];
  try { return cleanScanZones_(JSON.parse(raw)) || []; } catch (e) { return []; }
}
function saveScanZones_(req) {
  var zones = cleanScanZones_(req.zones);
  if (!zones) return { ok: false, error: 'Bad zones' };
  PropertiesService.getScriptProperties().setProperty('SCAN_ZONES', JSON.stringify(zones));
  return { ok: true, saved: true, zones: zones };
}
function cleanScanZones_(raw) {
  if (Object.prototype.toString.call(raw) !== '[object Array]') return null;
  var out = [], seenJo = false, seenDate = false, nOther = 0, i, z, id, x, y, w, h;
  for (i = 0; i < raw.length && out.length < 8; i++) {
    z = raw[i] || {};
    id = String(z.id || '');
    x = Number(z.x); y = Number(z.y); w = Number(z.w); h = Number(z.h);
    if (!(x >= 0 && y >= 0 && w > 0 && h > 0)) continue;
    if (x > 1 || y > 1 || w > 1 || h > 1) continue;
    if (x + w > 1) w = 1 - x;
    if (y + h > 1) h = 1 - y;
    if (w < 0.02 || h < 0.02) continue;
    if (id === 'jo') { if (seenJo) continue; seenJo = true; }
    else if (id === 'date') { if (seenDate) continue; seenDate = true; }
    else if (id === 'other' || /^other[2-4]$/.test(id)) { nOther++; if (nOther > 4) continue; id = nOther === 1 ? 'other' : 'other' + nOther; }
    else continue;
    out.push({ id: id, x: Math.round(x * 10000) / 10000, y: Math.round(y * 10000) / 10000, w: Math.round(w * 10000) / 10000, h: Math.round(h * 10000) / 10000 });
  }
  return out;
}
function ocrCrops_(req, filename) {
  if (!req.ocrBase64) return null;
  var joText = ocr_(Utilities.newBlob(Utilities.base64Decode(String(req.ocrBase64)), 'image/jpeg', 'jo-' + filename), 'jo-' + filename);
  var ocrText = joText, ocrError = '', extras = req.ocrFieldsBase64, i, ft;
  if (typeof extras === 'string' && extras) extras = [extras];
  if (Object.prototype.toString.call(extras) === '[object Array]') {
    for (i = 0; i < extras.length && i < 6; i++) {
      if (!extras[i]) continue;
      try {
        ft = ocr_(Utilities.newBlob(Utilities.base64Decode(String(extras[i])), 'image/jpeg', 'fld' + i + '-' + filename), 'fld' + i + '-' + filename);
        if (ft) ocrText += '\n' + ft;
      } catch (e2) { if (!ocrError) ocrError = String(e2); }
    }
  }
  return { joText: joText, ocrText: ocrText, ocrError: ocrError };
}
