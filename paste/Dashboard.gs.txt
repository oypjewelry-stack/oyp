Assumptions the live description did not spell out, so the diff can check them
- Sheet headers are trimmed, lowercased, and spaces become underscores before a row is read.
- A card's since is the Manila date of the JobStages row that is 'doing' for that JO and stage (started_at, else since, else updated_at).
- A card's client is the job's own client text. When can.clients, a Clients row matched on client_id replaces it with that row's name.
- An unknown location value is counted as unset.
- FORBIDDEN is { ok:false, error:'Not allowed', code:'FORBIDDEN' }.
- Confirmed payments are status confirmed. Rejected payments are left out of the recent list.

Helpers this file calls and does not define: ss_, isExec_, stripPricing_, and, when they exist, jobKey_, jobMergeVersions_, dashInquiries_.

function dashboard_(req, user) {
  var role = user && user.role;
  if (!DASH_ROLES[role]) return { ok: false, error: 'Not allowed', code: 'FORBIDDEN' };
  var exec = dashExec_(user);
  var hit = dashSheetGet_(role, exec);
  if (hit) return hit;

  try { lock = LockService.getScriptLock(); locked = lock.tryLock(8000); } catch (e0) { locked = false; }
  try {

    var fresh = dashBuild_(req, user);
    if (fresh && fresh.ok) dashSheetPut_(role, exec, fresh);
    return fresh;
  } finally {
    if (locked) { try { lock.releaseLock(); } catch (e1) {} }
  }
}

/** Still-out rows on the one Consignment tab. Does not read Consignees, Consignments, ConsignmentItems, or Items. */
function dashConsign_(today, limit, exec, liveJobs) {
  function looseCount(onSheet) {
    var n = 0;
    (liveJobs || []).forEach(function (j) {
      var k = dashJo_(j.jo);
      if (!k || (onSheet && onSheet[k])) return;
      if (String(j.location || '').trim().toLowerCase() !== 'consigned') return;
      n++;
    });
    return n;
  }
  var empty = { missing: [], consignments: [], counts: { consigned_out: 0, overdue_consignments: 0, consigned_jos: 0, owed_by_consignees: 0 } };
  if (!ss_().getSheetByName('Consignment')) {
    empty.missing = ['Consignment'];
    empty.counts.consigned_jos = looseCount(null);
    return empty;
  }
  var list = [], onSheet = {}, owed = 0;
  dashRead_('Consignment', []).forEach(function (r) {
    var st = String(r.status || '').trim().toLowerCase();
    var jo = dashJo_(r.jo);
    if (st === 'sold' && exec) {
      var agreed = dashNum_(r.agreed_value != null && String(r.agreed_value).trim() !== '' ? r.agreed_value : r.value);
      if (agreed != null) {
        var paid = dashNum_(r.paid_amount);
        if (paid == null) paid = 0;
        owed = Math.round((owed + Math.max(0, agreed - paid)) * 100) / 100;
      }
    }
    if (st !== '' && st !== 'consigned' && st !== 'open' && st !== 'out') return;
    var dout = dashDay_(r.date_out), due = dashDay_(r.due_back || r.due_date);
    var days = dashDaysBetween_(dout, today);
    var late = due ? today > due : (days != null && days > limit);
    var row = {
      description: String(r.description || ''), item_id: String(r.item_id || ''), jo: jo,
      consignee: String(r.consignee_name || r.consignee || ''), consignment_id: String(r.consignment_id || ''),
      date_out: dout, days_out: days, overdue: !!late
    };
    var value = dashNum_(r.agreed_value != null && String(r.agreed_value).trim() !== '' ? r.agreed_value : r.value);
    if (exec && value != null) row.value = value;
    if (jo) onSheet[jo] = 1;
    list.push(row);
  });
  list.sort(function (a, b) { return String(a.date_out).localeCompare(String(b.date_out)) || String(a.consignment_id).localeCompare(String(b.consignment_id)); });
  var overdueN = 0;
  list.forEach(function (r) { if (r.overdue) overdueN++; });
  var loose = looseCount(onSheet);
  return {
    missing: [], consignments: list,
    counts: { consigned_out: list.length, overdue_consignments: overdueN, consigned_jos: loose, owed_by_consignees: owed }
  };
}

function dashRead_(name, missing) {
  var sh;
  try { sh = ss_().getSheetByName(name); } catch (e0) { sh = null; }
  if (!sh) { missing.push(name); return []; }
  var values;
  try { values = sh.getDataRange().getValues(); } catch (e1) { return []; }
  if (!values || values.length < 2) return [];
  var heads = values[0].map(function (h) { return String(h || '').trim().toLowerCase().replace(/\s+/g, '_'); });
  var rows = [], r, c, obj, blank;
  for (r = 1; r < values.length; r++) {
    obj = {}; blank = true;
    for (c = 0; c < heads.length; c++) {
      if (!heads[c]) continue;
      obj[heads[c]] = values[r][c];
      if (values[r][c] != null && String(values[r][c]).trim() !== '') blank = false;
    }
    if (!blank) rows.push(obj);
  }
  if (name === 'JobOrders' && typeof jobMergeVersions_ === 'function') {
    try { rows = jobMergeVersions_(rows) || rows; } catch (e2) {}
  }
  return rows;
}

function dashJo_(v) {

function dashClient_(j, byId, on) {
  var raw = String(j.client || j.client_name || '').trim();
  if (!on) return raw;
  var id = String(j.client_id || '').trim();
  var found = (id && byId[id]) || (raw && byId[raw]);
  if (!found) return raw;
  return String(found.name || found.client_name || found.client || raw).trim();
}
function dashToday_() {
  try { return Utilities.formatDate(new Date(), 'Asia/Manila', 'yyyy-MM-dd'); } catch (e) {}
  return new Date(Date.now() + DASH.TZ_OFFSET_H * 36e5).toISOString().slice(0, 10);
}
function dashDay_(d) {
  if (d == null || d === '') return '';
  if (Object.prototype.toString.call(d) === '[object Date]' && !isNaN(d.getTime())) {
    try { return Utilities.formatDate(d, 'Asia/Manila', 'yyyy-MM-dd'); } catch (e) {}
  }
  var s = String(d);
  var m = s.match(/(\d{4}-\d{2}-\d{2})/);
  if (m) return m[1];
  var t = new Date(s);
  if (!isNaN(t.getTime())) {
    try { return Utilities.formatDate(t, 'Asia/Manila', 'yyyy-MM-dd'); } catch (e2) {}
  }
  return '';
}
function dashDaysBetween_(a, b) {
  if (!a || !b) return null;
  var x = Date.parse(a + 'T00:00:00Z'), y = Date.parse(b + 'T00:00:00Z');
  if (isNaN(x) || isNaN(y)) return null;
  return Math.round((y - x) / 86400000);
}
function dashNum_(v) {
  if (v == null || String(v).trim() === '') return null;
  var n = Number(String(v).replace(/[₱,\s]/g, '').replace(/php/ig, ''));
  return isFinite(n) ? Math.round(n * 100) / 100 : null;
}
function dashAddMonth_(ym, delta) {
  var p = String(ym).split('-'), y = Number(p[0]), m = Number(p[1]) - 1 + delta;
  y += Math.floor(m / 12);
  m = ((m % 12) + 12) % 12;
  return y + '-' + (m < 9 ? '0' : '') + (m + 1);
}

/** Same generation key as DashCache.gs, so a write there drops this copy too. 15 minutes. Per role and executive flag. */
function dashSheetKey_(role, exec) {
  var gen = '';
  try { gen = CacheService.getScriptCache().get('oypDashGen') || ''; } catch (e) { return ''; }
  if (!gen) return '';
  return 'd1:' + gen + ':' + role + ':' + (exec ? '1' : '0') + ':dashboard:-:-';
}
function dashSheetGet_(role, exec) {
  var head = dashSheetKey_(role, exec);
  if (!head) return null;
  var cache = CacheService.getScriptCache(), meta, info, parts = [], i, raw;
  try { meta = cache.get(head); } catch (e0) { return null; }
  if (!meta) return null;
  try { info = JSON.parse(meta); } catch (e1) { return null; }
  if (!info || !info.n) return null;
  for (i = 0; i < info.n; i++) {
    try { raw = cache.get(head + ':p' + i); } catch (e2) { return null; }
    if (raw == null) return null;
    parts.push(raw);
  }
  try { return JSON.parse(parts.join('')); } catch (e3) { return null; }
}
function dashSheetPut_(role, exec, value) {
  var cache = CacheService.getScriptCache(), gen = '';
  try { gen = cache.get('oypDashGen') || ''; } catch (e0) { gen = ''; }
  if (!gen) {
    gen = String(Date.now());
    try { cache.put('oypDashGen', gen, 21600); } catch (e1) { return; }
  }
  var head = 'd1:' + gen + ':' + role + ':' + (exec ? '1' : '0') + ':dashboard:-:-';
  var text;
  try { text = JSON.stringify(value); } catch (e2) { return; }
  var parts = [], i = 0, piece;
  while (i < text.length && parts.length < 8) {
    piece = text.slice(i, i + 80000);
    while (piece.length > 1000 && dashUtf8_(piece) > 90000) piece = piece.slice(0, Math.floor(piece.length * 0.9));
    parts.push(piece);
    i += piece.length;
  }
  if (i < text.length) return;
  try {
    cache.put(head, JSON.stringify({ n: parts.length }), 900);
    for (i = 0; i < parts.length; i++) cache.put(head + ':p' + i, parts[i], 900);
    if (typeof dashRemember_ === 'function') dashRemember_(gen, head);
  } catch (e3) {}
}
function dashUtf8_(s) {
  try { return Utilities.newBlob(String(s)).getBytes().length; } catch (e) { return String(s).length; }
}
