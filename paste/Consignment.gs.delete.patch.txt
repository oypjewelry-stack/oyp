Consignment.gs — an admin can delete one consignment sheet.
Do not paste a new Consignment.gs over the live file. Apply each FIND / REPLACE in order.
No new OAuth scopes. Do not add a column. Do not add consignDelete to FAST.READS, FAST.CACHE, or DCACHE.READS. It is a write.

Closing a sheet is unchanged: when the last piece is returned or sold, consignItem_ still records consignClose and the rows stay. Delete is separate. It is only for role admin.

Pieces whose status is still consigned go back to prev_location before the rows are removed, unless another sheet still has that JO out. A returned or sold piece is left where it already is.

If a tab named Deleted, AuditLog, or Audit already exists, one row is appended there with the deleted rows as JSON. The tab is not created. When none of those tabs exist, the response includes rows so the page can offer Undo for about 10 seconds.

function consignLogDeleted_(by, cid, rows) {
  try {

function consignDelete_(req, user) {
  if (!user || String(user.role || '') !== 'admin') return { ok: false, error: 'Deleting a sheet is for an admin', code: 'FORBIDDEN' };
  if (req && req.restore) return consignRestore_(req, user);
  var miss = consignMissing_(), bad = consignHeadersOk_();
  if (miss.length) return { ok: false, error: 'Run setupConsignment first' };
  if (bad) return { ok: false, error: bad };
  var cid = jobStr_(req.consignment_id);
  if (!cid) return { ok: false, error: 'Name the sheet' };
  var rows = [], skip = {}, i, r, jo, back, job, seenJo = {}, moved = [], copy = [], nums = [];
  consignRows_().forEach(function (row) {
    if (jobStr_(consignCell_(row, ['consignment_id'])) === cid) rows.push(row);
  });
  if (!rows.length) return { ok: false, error: 'No such sheet' };
  for (i = 0; i < rows.length; i++) skip[consignLineId_(rows[i])] = true;
  var t = now_(), by = user.username;
  for (i = 0; i < rows.length; i++) {
    r = rows[i];
    if (consignNormStatus_(consignCell_(r, ['status'])) !== 'consigned') continue;
    jo = jobKey_(consignCell_(r, ['jo']));
    if (!jo || seenJo[jo]) continue;
    seenJo[jo] = true;
    if (consignOutBesides_(jo, skip)) continue;
    back = consignLoc_(consignCell_(r, ['prev_location', 'previous_location'])) || 'vault';
    if (!back || back === 'consigned') back = 'vault';
    job = consignJob_(jo);
    if (job) { consignSetLoc_(job, back, user, t); moved.push(jo + ' -> ' + back); }
  }
  for (i = 0; i < rows.length; i++) copy.push(consignPlain_(rows[i]));
  var sh = ss_().getSheetByName(CONSIGN_TAB);
  if (!sh || !sh.deleteRow) return { ok: false, error: 'The Consignment tab cannot remove a row' };
  for (i = 0; i < rows.length; i++) if (rows[i]._row > 1) nums.push(rows[i]._row);
  nums.sort(function (a, b) { return b - a; });
  for (i = 0; i < nums.length; i++) sh.deleteRow(nums[i]);
  var logged = consignLogDeleted_(by, cid, copy);
  try { audit_(by, 'consignDelete', cid, req.device_id || '', copy.length + ' row(s)' + (moved.length ? ' · ' + moved.join('; ') : '')); } catch (eLog) {}
  var res = { ok: true, consignment_id: cid, removed: copy.length, consignee: jobStr_(consignCell_(rows[0], ['consignee_name', 'consignee'])), logged: logged };
  if (!logged) res.rows = copy;
  return res;
}
