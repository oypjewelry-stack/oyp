26-380 and 26-380B are different jobs. pfKey_ must not call jobKey_ or normJO_. Those drop a trailing letter, so a card named 26-380B-2a.jpg never matches key 26-380. The key stays exact: 26-380 matches only 26-380-<page><a|b>, and 26-380B matches only 26-380B-<page><a|b>.

  return String(s || '').trim().toUpperCase();

function pfKey_(s) {
  var t = String(s || '').trim().toUpperCase().replace(/\s+/g, '');
  var m = t.match(/^(\d{2})-(\d{3,4})([A-Z])?$/);
  if (m) return m[1] + '-' + m[2] + (m[3] || '');
  return t;
}

function pfCardParts_(joKey, fileName) {
  var base = String(fileName || '').replace(/\.[^.]+$/, '');
  var m = base.match(new RegExp('^' + pfEsc_(joKey) + '-(\\d+)([ab])$', 'i'));
  if (!m) return null;
  var page = parseInt(m[1], 10);
  if (!(page > 0)) return null;
  return { page: page, side: m[2].toLowerCase() };
}
