/**
 * Consignment sheets: pieces lent to someone outside the shop (a relative, a consignee).
 * Web actions (admin + production manager): consignList, consignSearch, consignSave, consignItem, consigneeSave.
 * One-time: run setupConsignment. It adds Consignees, Consignments, ConsignmentItems only when that tab is missing.
 * Never deletes a row. Never edits a tab that already exists.
 *
 * Auth.gs handleAction_, immediately after the JobSync line:
 *   if (typeof isConsignAction_ === 'function' && isConsignAction_(a)) return handleConsign_(req, s.user); // Consignment.gs
 */
var CONSIGN_TABS = {
  Consignees: ['consignee_id', 'name', 'phone', 'notes', 'created_at', 'created_by', 'updated_at', 'updated_by'],
  Consignments: ['consignment_id', 'consignee_id', 'consignee_name', 'date_out', 'due_date', 'status', 'notes', 'created_at', 'created_by', 'updated_at', 'updated_by', 'closed_at'],
  ConsignmentItems: ['line_id', 'consignment_id', 'consignee', 'item_id', 'jo', 'description', 'value', 'status', 'date_out', 'due_date', 'prev_location', 'returned_at', 'sold_at', 'created_at', 'created_by']
};
var CONSIGN_ROLES = { admin: true, production_manager: true };

/** ONE-TIME. Adds the three tabs with headers only if the tab is missing. Does not change a tab that exists. */
function setupConsignment() {
  var run = function () {
    var ss = ss_(), added = [], kept = [], name, sh, cols;
    for (name in CONSIGN_TABS) if (CONSIGN_TABS.hasOwnProperty(name)) {
      cols = CONSIGN_TABS[name];
      sh = ss.getSheetByName(name);
      if (sh) { kept.push(name); continue; }
      sh = ss.insertSheet(name);
      sh.getRange(1, 1, 1, cols.length).setValues([cols]);
      sh.setFrozenRows(1);
      added.push(name);
    }
    return { added: added, kept: kept, url: ss.getUrl() };
  };
  var res = (typeof withLock_ === 'function') ? withLock_(run) : run();
  try { audit_('setup', 'setupConsignment', '', '', 'added ' + (res.added.join(', ') || 'none') + '; kept ' + (res.kept.join(', ') || 'none')); } catch (e) {}
  Logger.log('Added: ' + (res.added.join(', ') || 'none') + '\nLeft as they were: ' + (res.kept.join(', ') || 'none') + '\n' + res.url);
  return res;
}

function isConsignAction_(a) {
  return { consignList: 1, consignSearch: 1, consignSave: 1, consignItem: 1, consigneeSave: 1 }[String(a || '')] === 1;
}
function handleConsign_(req, user) {
  if (!user || !user.username) {
    try {
      var sess = (typeof sessionFrom_ === 'function') ? sessionFrom_(req || {}) : null;
      if (sess && sess.user) user = sess.user;
      else if (sess && sess.username) user = sess;
    } catch (e) {}
  }
  var acts = { consignList: consignList_, consignSearch: consignSearch_, consignSave: consignSave_, consignItem: consignItem_, consigneeSave: consigneeSave_ };
  if (!req || !acts[req.action]) return { ok: false, error: 'Unknown action' };
  if (!user || !CONSIGN_ROLES[user.role]) {
    try { audit_((user && user.username) || '', 'denied', req.action, req.device_id, 'role ' + ((user && user.role) || '')); } catch (e2) {}
    return { ok: false, error: 'Not allowed', code: 'FORBIDDEN' };
  }
  if (!isExec_(user) && consignSentValue_(req)) {
    try { audit_(user.username, 'denied', req.action, req.device_id, 'agreed value from role ' + user.role); } catch (e3) {}
    return { ok: false, error: 'Agreed value is for executives only', code: 'FORBIDDEN' };
  }
  var write = req.action === 'consignSave' || req.action === 'consignItem' || req.action === 'consigneeSave';
  var res;
  try { res = write ? withLock_(function () { return acts[req.action](req, user); }) : acts[req.action](req, user); }
  catch (err) { res = { ok: false, error: String(err && err.message || err).slice(0, 300) }; }
  return (typeof stripPricing_ === 'function') ? stripPricing_(res, user) : res;
}
function consignSentValue_(req) {
  var items = req.items || [], i, v;
  for (i = 0; i < items.length; i++) {
    v = items[i] && (items[i].value != null ? items[i].value : items[i].agreed_value);
    if (String(v == null ? '' : v).trim() !== '') return true;
  }
  return false;
}

function consignSearch_(req, user) {
  var q = String(req.q || '').trim(), uq = q.toUpperCase().replace(/\s+/g, ' '), joQ = jobKey_(q);
  if (uq.length < 2 && !joQ) return { ok: true, q: q, hits: [] };
  var hits = [], seen = {};
  if (ss_().getSheetByName('JobOrders')) jobRows_('JobOrders').forEach(function (j) {
    if (jobStr_(j.status) === 'cancelled') return;
    var jo = jobKey_(j.jo), item = jobStr_(j.item_id), desc = jobStr_(j.description), loc = consignLoc_(j.location);
    if (!jo) return;
    var hay = (jo + ' ' + item + ' ' + desc).toUpperCase();
    if (!(joQ && jo.indexOf(joQ) >= 0) && hay.indexOf(uq) < 0 && (!item || item.toUpperCase().indexOf(uq) < 0)) return;
    consignPushHit_(hits, seen, { jo: jo, item_id: item, description: desc, location: loc });
  });
  if (ss_().getSheetByName('Items')) jobRows_('Items').forEach(function (r) {
    var item = jobStr_(consignCell_(r, ['item_id', 'sku', 'code'])), jo = jobKey_(consignCell_(r, ['jo', 'job', 'jo_number'])), desc = jobStr_(consignCell_(r, ['description', 'name', 'title'])), loc = consignLoc_(consignCell_(r, ['location']));
    if (!item && !jo) return;
    var hay = (jo + ' ' + item + ' ' + desc).toUpperCase();
    if (!(joQ && jo && jo.indexOf(joQ) >= 0) && hay.indexOf(uq) < 0) return;
    consignPushHit_(hits, seen, { jo: jo, item_id: item, description: desc, location: loc });
  });
  hits.forEach(function (h) { h.blocked = consignBlockReason_(h.jo, h.item_id, h.location); });
  hits.sort(function (a, b) { return consignCmp_(a.jo || a.item_id, b.jo || b.item_id); });
  return { ok: true, q: q, hits: hits.slice(0, 30) };
}

function consignSave_(req, user) {
  var miss = consignMissing_(), bad = consignHeadersOk_();
  if (miss.length) return { ok: false, error: 'Run setupConsignment first' };
  if (bad) return { ok: false, error: bad };
  var items = req.items;
  if (!Array.isArray(items) || !items.length) return { ok: false, error: 'Add at least one piece' };
  if (items.length > 40) return { ok: false, error: 'At most 40 pieces on a sheet' };
  var today = consignToday_(), limit = consignOverdueDays_(), due = jobStr_(req.due_date);
  if (due && !/^\d{4}-\d{2}-\d{2}$/.test(due)) return { ok: false, error: 'Due date must look like 2026-11-19' };
  if (!due) due = consignAddDays_(today, limit);
  var who = consignResolvePerson_(req, user);
  if (who.error) return who;
  var prepared = [], seenJo = {}, i, it, jo, rawJo, itemId, desc, val, job, loc, block;
  for (i = 0; i < items.length; i++) {
    it = items[i] || {};
    rawJo = jobStr_(it.jo); jo = rawJo ? jobKey_(rawJo) : '';
    if (rawJo && !jo) return { ok: false, error: 'JO number does not look right' };
    itemId = consignClean_(it.item_id, 40);
    if (!jo && !itemId) return { ok: false, error: 'Each piece needs a JO number or an item id' };
    if (jo && seenJo[jo]) return { ok: false, error: 'JO ' + jo + ' is on this sheet twice' };
    if (jo) seenJo[jo] = true;
    job = jo ? consignJob_(jo) : null;
    if (jo && !job) return { ok: false, error: 'No JO ' + jo };
    loc = job ? consignLoc_(job.location) : consignLoc_(it.location);
    block = consignBlockReason_(jo, itemId, loc);
    if (block) return { ok: false, error: block };
    desc = consignClean_(it.description, 300) || (job ? jobStr_(job.description) : '');
    if (!itemId && job) itemId = jobStr_(job.item_id);
    try { val = isExec_(user) ? consignMoney_(it.value != null ? it.value : it.agreed_value) : ''; }
    catch (e) { return { ok: false, error: e.message }; }
    prepared.push({ jo: jo, item_id: itemId, description: desc, value: val, prev: (loc && loc !== 'consigned') ? loc : 'vault', row: job });
  }
  var t = now_(), by = user.username, cid = consignNextId_(), notes = consignClean_(req.notes, 400);
  consignAppend_('Consignments', {
    consignment_id: cid, id: cid, consignee_id: who.id, consignee_name: who.name, consignee: who.name,
    date_out: today, due_date: due, status: 'open', notes: notes, created_at: t, created_by: by, updated_at: t, updated_by: by, closed_at: ''
  });
  var moved = [];
  prepared.forEach(function (p) {
    var lid = 'cl-' + randHex_(8);
    consignAppend_('ConsignmentItems', {
      line_id: lid, id: lid, consignment_id: cid, consignee: who.name, consignee_name: who.name,
      item_id: p.item_id, jo: p.jo, description: p.description, value: p.value, agreed_value: p.value,
      status: 'consigned', date_out: today, due_date: due, prev_location: p.prev, returned_at: '', sold_at: '',
      created_at: t, created_by: by
    });
    if (p.jo && !consignOthersOut_(p.jo, lid)) {
      consignSetLoc_(p.row, 'consigned', user, t);
      moved.push(p.jo + ' ' + p.prev + ' -> consigned');
    }
  });
  audit_(by, 'consignSave', cid, req.device_id, who.name + ' · ' + prepared.length + ' piece(s)' + (moved.length ? ' · ' + moved.join('; ') : '') + (isExec_(user) ? ' · values set' : ''));
  return { ok: true, consignment_id: cid };
}
function consignItem_(req, user) {
  var miss = consignMissing_(), bad = consignHeadersOk_();
  if (miss.length) return { ok: false, error: 'Run setupConsignment first' };
  if (bad) return { ok: false, error: bad };
  var status = String(req.status || '').toLowerCase();
  if (status !== 'returned' && status !== 'sold') return { ok: false, error: 'Status must be returned or sold' };
  var row = null, lineId = jobStr_(req.line_id);
  if (lineId) row = consignFind_('ConsignmentItems', ['line_id', 'id'], lineId);
  else row = consignMatchCode_(jobStr_(req.code), jobStr_(req.consignment_id));
  if (row && row.error) return row;
  if (!row) return { ok: false, error: lineId ? 'No such line' : 'No matching piece is still out' };
  lineId = jobStr_(consignCell_(row, ['line_id', 'id']));
  if (!consignItemOpen_(consignCell_(row, ['status']))) return { ok: false, error: 'That piece is already ' + jobStr_(consignCell_(row, ['status'])) };
  var t = now_(), by = user.username, jo = jobKey_(consignCell_(row, ['jo'])), cid = jobStr_(consignCell_(row, ['consignment_id']));
  var changes = { status: status, updated_at: t, updated_by: by };
  if (status === 'returned') changes.returned_at = t; else changes.sold_at = t;
  consignUpdate_('ConsignmentItems', row._row, changes);
  var locNote = '';
  if (jo && !consignOthersOut_(jo, lineId)) {
    var back = status === 'sold' ? 'sold' : (consignLoc_(consignCell_(row, ['prev_location', 'previous_location'])) || 'vault');
    if (back === 'consigned') back = 'vault';
    var job = consignJob_(jo);
    if (job) { consignSetLoc_(job, back, user, t); locNote = jo + ' -> ' + back; }
  } else if (jo) locNote = jo + ' still has another piece out';
  consignMaybeClose_(cid, user, t);
  audit_(by, status === 'sold' ? 'consignSold' : 'consignReturn', lineId, req.device_id, cid + (locNote ? ' · ' + locNote : ''));
  return { ok: true, line_id: lineId, consignment_id: cid, status: status };
}

/* ---------- sheet building ---------- */
function consignSheets_(people, today, limit) {
  var byId = {}; people.forEach(function (p) { byId[p.consignee_id] = p; });
  var items = {}, order = [];
  jobRows_('ConsignmentItems').forEach(function (r) {
    var cid = jobStr_(consignCell_(r, ['consignment_id'])); if (!cid) return;
    if (!items[cid]) { items[cid] = []; order.push(cid); }
    items[cid].push(r);
  });
  var out = [], seen = {};
  jobRows_('Consignments').forEach(function (sh) {
    var cid = jobStr_(consignCell_(sh, ['consignment_id', 'id'])); if (!cid || seen[cid]) return; seen[cid] = true;
    out.push(consignSheetView_(sh, items[cid] || [], byId, today, limit));
  });
  order.forEach(function (cid) { if (!seen[cid]) out.push(consignSheetView_({ consignment_id: cid }, items[cid], byId, today, limit)); });
  out.sort(function (a, b) { return consignCmp_(a.date_out, b.date_out) || consignCmp_(a.consignment_id, b.consignment_id); });
  return out;
}
function consignSheetView_(sh, rows, byId, today, limit) {
  var gid = jobStr_(consignCell_(sh, ['consignee_id'])), person = byId[gid];
  var name = person ? person.name : jobStr_(consignCell_(sh, ['consignee_name', 'consignee', 'name']));
  var dateOut = consignDay_(consignCell_(sh, ['date_out', 'date'])) || consignDay_(rows[0] && consignCell_(rows[0], ['date_out']));
  var due = consignDay_(consignCell_(sh, ['due_date', 'due']));
  var viewItems = rows.map(function (r) {
    var st = consignNormStatus_(consignCell_(r, ['status']));
    var dout = consignDay_(consignCell_(r, ['date_out'])) || dateOut;
    var idue = consignDay_(consignCell_(r, ['due_date'])) || due;
    var open = st === 'consigned';
    var o = {
      line_id: jobStr_(consignCell_(r, ['line_id', 'id'])), item_id: jobStr_(consignCell_(r, ['item_id', 'sku', 'code'])),
      jo: jobKey_(consignCell_(r, ['jo'])) || jobStr_(consignCell_(r, ['jo'])), description: jobStr_(consignCell_(r, ['description', 'name'])),
      status: st, date_out: dout, due_date: idue, prev_location: consignLoc_(consignCell_(r, ['prev_location', 'previous_location'])),
      returned_at: jobStr_(consignCell_(r, ['returned_at'])), sold_at: jobStr_(consignCell_(r, ['sold_at'])),
      days_out: consignSpan_(dout, today), overdue: open && consignIsOverdue_(dout, idue, today, limit)
    };
    var val = consignCell_(r, ['value', 'agreed_value']);
    if (val !== '' && val != null) o.value = consignMoneySafe_(val);
    return o;
  });
  var anyOpen = viewItems.some(function (it) { return it.status === 'consigned'; });
  var stored = jobStr_(consignCell_(sh, ['status'])).toLowerCase();
  return {
    consignment_id: jobStr_(consignCell_(sh, ['consignment_id', 'id'])) || (rows[0] && jobStr_(consignCell_(rows[0], ['consignment_id']))),
    consignee_id: gid, consignee: name, phone: person ? person.phone : '', notes: jobStr_(consignCell_(sh, ['notes'])),
    date_out: dateOut, due_date: due, status: (!viewItems.length ? (stored === 'closed' ? 'closed' : 'open') : (anyOpen ? 'open' : 'closed')),
    days_out: consignSpan_(dateOut, today), overdue: anyOpen && viewItems.some(function (it) { return it.overdue; }), items: viewItems
  };
}
function consignLoose_(sheets) {
  var outJo = {};
  sheets.forEach(function (s) { (s.items || []).forEach(function (it) { if (it.status === 'consigned' && it.jo) outJo[jobKey_(it.jo)] = true; }); });
  var loose = [];
  if (!ss_().getSheetByName('JobOrders')) return loose;
  jobRows_('JobOrders').forEach(function (j) {
    var k = jobKey_(j.jo);
    if (!k || outJo[k] || consignLoc_(j.location) !== 'consigned' || jobStr_(j.status) === 'cancelled') return;
    loose.push({ jo: k, item_id: jobStr_(j.item_id), description: jobStr_(j.description), location: 'consigned' });
  });
  loose.sort(function (a, b) { return consignCmp_(a.jo, b.jo); });
  return loose;
}
function consignPeople_() {

  }).filter(function (p) { return p.consignee_id && p.name; });

}

  if (!consignHas_('ConsignmentItems', ['consignment_id']) || !consignHas_('ConsignmentItems', ['status']) || !consignHas_('ConsignmentItems', ['line_id', 'id'])) return 'ConsignmentItems needs line_id, consignment_id, and status columns. An existing tab is not changed.';
  return '';
}
function consignMissing_() {
  var ss = ss_(), miss = [], name;

}
function consignCell_(row, keys) {
  var i, v; if (!row) return '';
  for (i = 0; i < keys.length; i++) { v = row[keys[i]]; if (v != null && String(v).trim() !== '') return v; }
  return '';
}

function consignResolvePerson_(req, user) {
  var id = jobStr_(req.consignee_id), t = now_();
  if (id) {
    var row = consignFind_('Consignees', ['consignee_id', 'id'], id);
    if (!row) return { error: 'Pick a consignee', ok: false };
    return { id: id, name: jobStr_(consignCell_(row, ['name', 'consignee_name', 'consignee'])) };

function consignMatchCode_(code, onlySheet) {
  var jo = jobKey_(code), raw = String(code || '').trim().toLowerCase(), hits = [];
  if (!jo && !raw) return null;
  jobRows_('ConsignmentItems').forEach(function (r) {
    if (!consignItemOpen_(consignCell_(r, ['status']))) return;
    var cid = jobStr_(consignCell_(r, ['consignment_id']));
    if (onlySheet && cid !== onlySheet) return;
    var rjo = jobKey_(consignCell_(r, ['jo'])), item = jobStr_(consignCell_(r, ['item_id', 'sku', 'code'])).toLowerCase();
    if ((jo && rjo === jo) || (raw && item && item === raw)) hits.push(r);
  });
  if (!hits.length && onlySheet) return consignMatchCode_(code, '');
  if (hits.length > 1) return { ok: false, error: 'More than one piece matches. Tap the row.' };
  return hits[0] || null;
}
function consignBlockReason_(jo, itemId, loc) {
  if (loc === 'sold') return (jo || itemId || 'This piece') + ' is already sold';
  if (loc === 'archived') return (jo || itemId || 'This piece') + ' is archived';
  var key = jobKey_(jo), id = jobStr_(itemId).toLowerCase(), hit = false;
  if (!ss_().getSheetByName('ConsignmentItems')) return '';
  jobRows_('ConsignmentItems').forEach(function (r) {
    if (hit || !consignItemOpen_(consignCell_(r, ['status']))) return;
    if (key && jobKey_(consignCell_(r, ['jo'])) === key) hit = true;
    else if (!key && id && jobStr_(consignCell_(r, ['item_id', 'sku', 'code'])).toLowerCase() === id) hit = true;
  });
  return hit ? (key || itemId) + ' is already on an open sheet' : '';
}

function consignSpan_(from, to) {
  if (!from || !to) return null;
  var a = Date.parse(from + 'T00:00:00Z'), b = Date.parse(to + 'T00:00:00Z');
  if (isNaN(a) || isNaN(b)) return null;
  return Math.round((b - a) / 86400000);
}
