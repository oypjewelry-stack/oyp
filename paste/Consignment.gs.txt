This file only uses the spreadsheet helpers the project already has (ss_, jobRows_, jobAppend_, jobSet_, jobHeader_, jobKey_, jobStr_, jobDay_, withLock_, audit_, now_, randHex_, isExec_, stripPricing_). sessionFrom_ already runs in Auth.gs before the hook below; handleConsign_ calls it only if that hook did not pass a user. The camera stays in the browser. Nothing here calls Drive, Gmail, or UrlFetch.

Dashboard.gs already reads Consignees, Consignments, ConsignmentItems, and Items, and the Settings key consign_overdue_days (default 30). This file writes those same columns. It never deletes a row. setupConsignment adds the three tabs with a header row only when the tab is missing. A tab that already exists is not rewritten and not resized. The one exception: if ConsignmentItems is already there, setupConsignment adds the header cells owed, paid_amount, and paid_at at the far right of row 1 when those names are missing. It does not change any other header cell and it does not change data rows.

Headers created when the tab is missing:
- Consignees: consignee_id, name, phone, notes, created_at, created_by, updated_at, updated_by
- Consignments: consignment_id, consignee_id, consignee_name, date_out, due_date, status, notes, created_at, created_by, updated_at, updated_by, closed_at
- ConsignmentItems: line_id, consignment_id, consignee, item_id, jo, description, value, status, date_out, due_date, prev_location, returned_at, sold_at, owed, paid_amount, paid_at, created_at, created_by

If a tab already exists, writes fill whichever of those header names are already on row 1 (value is also written to agreed_value when that header exists; the line id is also written to id when that header exists). A missing header is left missing, except the three money headers above, which setupConsignment can add.

When a piece is marked sold, the consignee owes OYP the agreed value. owed is copied from that value at the moment of the sale. paid_amount is the cumulative amount recorded against the line, and paid_at is the date of the latest payment. A returned piece does not need a payment. A sheet closes only when every piece is returned, or sold and paid (paid_amount covers owed). A blank agreed value is not treated as paid.

Payments.gs is not in this project copy, and paySave is not a safe function to call from here: the payments page requires a JO plus a receipt photo or a reference, and inventing a Payments row with guessed columns can break Quotes and payments. This file does not write a Payments row. Recording a payment stores paid_amount and paid_at on the ConsignmentItems line and audits it. A payment recorded here counts immediately (status confirmed), the same rule as a manual payment on the payments page. Pending is only for payments that arrive by themselves (Gmail, Telegram). The audit line is the history: payer is the consignee, reference is the consignment id plus the line id (ci_id).

Agreed value, owed, paid, and balance are executive-only. isExec_ / stripPricing_ cover value. owed, paid, and balance are left off the response for anyone who is not an executive, because those names are not in the pricing strip list. A production manager can consign, return, and sell, and does not receive the amounts or record a payment.

/**
 * Consignment sheets: pieces lent to someone outside the shop (a relative, a consignee).
 * Web actions (admin + production manager): consignList, consignSearch, consignSave, consignItem, consigneeSave.
 * consignPay is executive-only. It stores the payment on the consignment line (paid_amount, paid_at). It does not write Payments.
 * One-time: run setupConsignment. It adds Consignees, Consignments, ConsignmentItems only when that tab is missing.
 * Never deletes a row. An existing tab is not rewritten. ConsignmentItems may gain owed, paid_amount, paid_at at the far right.
 *
 * Auth.gs handleAction_, immediately after the JobSync line:
 *   if (typeof isConsignAction_ === 'function' && isConsignAction_(a)) return handleConsign_(req, s.user); // Consignment.gs
 */
var CONSIGN_TABS = {
  Consignees: ['consignee_id', 'name', 'phone', 'notes', 'created_at', 'created_by', 'updated_at', 'updated_by'],
  Consignments: ['consignment_id', 'consignee_id', 'consignee_name', 'date_out', 'due_date', 'status', 'notes', 'created_at', 'created_by', 'updated_at', 'updated_by', 'closed_at'],
  ConsignmentItems: ['line_id', 'consignment_id', 'consignee', 'item_id', 'jo', 'description', 'value', 'status', 'date_out', 'due_date', 'prev_location', 'returned_at', 'sold_at', 'owed', 'paid_amount', 'paid_at', 'created_at', 'created_by']
};
var CONSIGN_PAY_COLS = ['owed', 'paid_amount', 'paid_at'];
var CONSIGN_ROLES = { admin: true, production_manager: true };

/** ONE-TIME. Adds a missing tab with its header row. On an existing ConsignmentItems tab, adds owed, paid_amount, paid_at at the far right only when that name is absent. */
function setupConsignment() {
  var run = function () {
    var ss = ss_(), added = [], kept = [], colsAdded = [], name, sh, cols, head, i, next;
    for (name in CONSIGN_TABS) if (CONSIGN_TABS.hasOwnProperty(name)) {
      cols = CONSIGN_TABS[name];
      sh = ss.getSheetByName(name);
      if (sh) { kept.push(name); continue; }
      sh = ss.insertSheet(name);
      sh.getRange(1, 1, 1, cols.length).setValues([cols]);
      sh.setFrozenRows(1);
      added.push(name);
    }
    sh = ss.getSheetByName('ConsignmentItems');
    if (sh && sh.getLastRow() >= 1 && typeof jobHeader_ === 'function') {
      head = jobHeader_(sh);
      next = head.length + 1;
      for (i = 0; i < CONSIGN_PAY_COLS.length; i++) {
        if (head.indexOf(CONSIGN_PAY_COLS[i]) >= 0) continue;
        sh.getRange(1, next).setValue(CONSIGN_PAY_COLS[i]);
        colsAdded.push(CONSIGN_PAY_COLS[i]);
        head.push(CONSIGN_PAY_COLS[i]);
        next++;
      }
    }
    return { added: added, kept: kept, columns: colsAdded, url: ss.getUrl() };
  };
  var res = (typeof withLock_ === 'function') ? withLock_(run) : run();
  try { audit_('setup', 'setupConsignment', '', '', 'added ' + (res.added.join(', ') || 'none') + '; kept ' + (res.kept.join(', ') || 'none') + '; columns ' + ((res.columns || []).join(', ') || 'none')); } catch (e) {}
  Logger.log('Added: ' + (res.added.join(', ') || 'none') + '\nLeft as they were: ' + (res.kept.join(', ') || 'none') + '\nColumns added on ConsignmentItems: ' + ((res.columns || []).join(', ') || 'none') + '\n' + res.url);
  return res;
}

function isConsignAction_(a) {
  return { consignList: 1, consignSearch: 1, consignSave: 1, consignItem: 1, consigneeSave: 1, consignPay: 1 }[String(a || '')] === 1;
}
function handleConsign_(req, user) {
  if (!user || !user.username) {
    try {
      var sess = (typeof sessionFrom_ === 'function') ? sessionFrom_(req || {}) : null;
      if (sess && sess.user) user = sess.user;
      else if (sess && sess.username) user = sess;
    } catch (e) {}
  }
  var acts = { consignList: consignList_, consignSearch: consignSearch_, consignSave: consignSave_, consignItem: consignItem_, consigneeSave: consigneeSave_, consignPay: consignPay_ };
  if (!req || !acts[req.action]) return { ok: false, error: 'Unknown action' };
  if (!user || !CONSIGN_ROLES[user.role]) {
    try { audit_((user && user.username) || '', 'denied', req.action, req.device_id, 'role ' + ((user && user.role) || '')); } catch (e2) {}
    return { ok: false, error: 'Not allowed', code: 'FORBIDDEN' };
  }
  if (!isExec_(user) && consignSentValue_(req)) {
    try { audit_(user.username, 'denied', req.action, req.device_id, 'agreed value from role ' + user.role); } catch (e3) {}
    return { ok: false, error: 'Agreed value is for executives only', code: 'FORBIDDEN' };
  }
  if (req.action === 'consignPay' && !isExec_(user)) {
    try { audit_(user.username, 'denied', req.action, req.device_id, 'payment from role ' + user.role); } catch (e4) {}
    return { ok: false, error: 'Recording a payment is for executives only', code: 'FORBIDDEN' };
  }
  var write = req.action === 'consignSave' || req.action === 'consignItem' || req.action === 'consigneeSave' || req.action === 'consignPay';
  var res;
  try { res = write ? withLock_(function () { return acts[req.action](req, user); }) : acts[req.action](req, user); }
  catch (err) { res = { ok: false, error: String(err && err.message || err).slice(0, 300) }; }
  return (typeof stripPricing_ === 'function') ? stripPricing_(res, user) : res;
}
function consignSentValue_(req) {
  var items = req.items || [], i, v;
  for (i = 0; i < items.length; i++) {
    v = items[i] && (items[i].value != null ? items[i].value : items[i].agreed_value);
    if (String(v == null ? '' : v).trim() !== '') return true;
  }
  return false;
}

function consignList_(req, user) {
  var miss = consignMissing_(), today = consignToday_(), limit = consignOverdueDays_(), exec = isExec_(user), people = consignPeople_(), sheets = [];
  if (miss.length === 0) sheets = consignSheets_(people, today, limit, exec);
  var outPeople = people.map(function (p) {
    var o = { consignee_id: p.consignee_id, name: p.name, phone: p.phone, notes: p.notes };
    if (exec) { o.sold = 0; o.owed = 0; o.paid = 0; o.balance = 0; }
    return o;
  });
  var res = {
    ok: true, setup: miss.length === 0, missing: miss, overdue_days: limit, today: today, prices: exec,
    consignees: outPeople, sheets: sheets, loose: miss.length ? [] : consignLoose_(sheets)
  };
  if (exec) {
    var roll = { sold: 0, owed: 0, paid: 0, balance: 0 }, i, slot;
    sheets.forEach(function (s) {
      roll.sold += s.sold || 0;
      roll.owed = consignRound_(roll.owed + (s.owed || 0));
      roll.paid = consignRound_(roll.paid + (s.paid || 0));
      roll.balance = consignRound_(roll.balance + (s.balance || 0));
      slot = null;
      for (i = 0; i < outPeople.length; i++) if (outPeople[i].consignee_id && outPeople[i].consignee_id === s.consignee_id) slot = outPeople[i];
      if (!slot) for (i = 0; i < outPeople.length; i++) if (outPeople[i].name && outPeople[i].name === s.consignee) slot = outPeople[i];
      if (!slot) return;
      slot.sold += s.sold || 0;
      slot.owed = consignRound_(slot.owed + (s.owed || 0));
      slot.paid = consignRound_(slot.paid + (s.paid || 0));
      slot.balance = consignRound_(slot.balance + (s.balance || 0));
    });
    res.sold = roll.sold;
    res.sold_count = roll.sold;
    res.owed = roll.owed;
    res.paid = roll.paid;
    res.balance = roll.balance;
  }
  return res;
}
function consignSearch_(req, user) {
  var q = String(req.q || '').trim(), uq = q.toUpperCase().replace(/\s+/g, ' '), joQ = jobKey_(q);
  if (uq.length < 2 && !joQ) return { ok: true, q: q, hits: [] };
  var hits = [], seen = {};
  if (ss_().getSheetByName('JobOrders')) jobRows_('JobOrders').forEach(function (j) {
    if (jobStr_(j.status) === 'cancelled') return;
    var jo = jobKey_(j.jo), item = jobStr_(j.item_id), desc = jobStr_(j.description), loc = consignLoc_(j.location);
    if (!jo) return;
    var hay = (jo + ' ' + item + ' ' + desc).toUpperCase();
    if (!(joQ && jo.indexOf(joQ) >= 0) && hay.indexOf(uq) < 0 && (!item || item.toUpperCase().indexOf(uq) < 0)) return;
    consignPushHit_(hits, seen, { jo: jo, item_id: item, description: desc, location: loc });
  });
  if (ss_().getSheetByName('Items')) jobRows_('Items').forEach(function (r) {
    var item = jobStr_(consignCell_(r, ['item_id', 'sku', 'code'])), jo = jobKey_(consignCell_(r, ['jo', 'job', 'jo_number'])), desc = jobStr_(consignCell_(r, ['description', 'name', 'title'])), loc = consignLoc_(consignCell_(r, ['location']));
    if (!item && !jo) return;
    var hay = (jo + ' ' + item + ' ' + desc).toUpperCase();
    if (!(joQ && jo && jo.indexOf(joQ) >= 0) && hay.indexOf(uq) < 0) return;
    consignPushHit_(hits, seen, { jo: jo, item_id: item, description: desc, location: loc });
  });
  hits.forEach(function (h) { h.blocked = consignBlockReason_(h.jo, h.item_id, h.location); });
  hits.sort(function (a, b) { return consignCmp_(a.jo || a.item_id, b.jo || b.item_id); });
  return { ok: true, q: q, hits: hits.slice(0, 30) };
}

function consignItem_(req, user) {
  var miss = consignMissing_(), bad = consignHeadersOk_();
  if (miss.length) return { ok: false, error: 'Run setupConsignment first' };
  if (bad) return { ok: false, error: bad };
  var status = String(req.status || '').toLowerCase();
  if (status !== 'returned' && status !== 'sold') return { ok: false, error: 'Status must be returned or sold' };
  var row = null, lineId = jobStr_(req.line_id);
  if (lineId) row = consignFind_('ConsignmentItems', ['line_id', 'id'], lineId);
  else row = consignMatchCode_(jobStr_(req.code), jobStr_(req.consignment_id));
  if (row && row.error) return row;
  if (!row) return { ok: false, error: lineId ? 'No such line' : 'No matching piece is still out' };
  lineId = jobStr_(consignCell_(row, ['line_id', 'id']));
  if (!consignItemOpen_(consignCell_(row, ['status']))) return { ok: false, error: 'That piece is already ' + jobStr_(consignCell_(row, ['status'])) };
  var t = now_(), by = user.username, jo = jobKey_(consignCell_(row, ['jo'])), cid = jobStr_(consignCell_(row, ['consignment_id']));
  var changes = { status: status, updated_at: t, updated_by: by }, owed = null;
  if (status === 'returned') changes.returned_at = t;
  else {
    if (!consignHas_('ConsignmentItems', ['owed'])) return { ok: false, error: 'Run setupConsignment again. It adds an owed column on ConsignmentItems and does not change existing cells.' };
    owed = consignNum_(consignCell_(row, ['value', 'agreed_value']));
    changes.sold_at = t;
    changes.owed = owed == null ? '' : owed;
  }
  consignUpdate_('ConsignmentItems', row._row, changes);
  var locNote = '';
  if (jo && !consignOthersOut_(jo, lineId)) {
    var back = status === 'sold' ? 'sold' : (consignLoc_(consignCell_(row, ['prev_location', 'previous_location'])) || 'vault');
    if (back === 'consigned') back = 'vault';
    var job = consignJob_(jo);
    if (job) { consignSetLoc_(job, back, user, t); locNote = jo + ' -> ' + back; }
  } else if (jo) locNote = jo + ' still has another piece out';
  consignSyncClose_(cid, user, t);
  audit_(by, status === 'sold' ? 'consignSold' : 'consignReturn', lineId, req.device_id, cid + (status === 'sold' ? ' · owed ' + (owed == null ? 'blank' : owed) : '') + (locNote ? ' · ' + locNote : ''));
  return { ok: true, line_id: lineId, consignment_id: cid, status: status };
}
function consignPay_(req, user) {
  var miss = consignMissing_(), bad = consignHeadersOk_();
  if (miss.length) return { ok: false, error: 'Run setupConsignment first' };
  if (bad) return { ok: false, error: bad };
  if (!consignHas_('ConsignmentItems', ['owed']) || !consignHas_('ConsignmentItems', ['paid_amount']) || !consignHas_('ConsignmentItems', ['paid_at'])) {
    return { ok: false, error: 'Run setupConsignment again. It adds owed, paid_amount, and paid_at on ConsignmentItems and does not change existing cells.' };
  }
  var amount;
  try { amount = consignMoney_(req.amount); }
  catch (e) { return { ok: false, error: 'Type the amount paid' }; }
  if (amount === '' || !(amount > 0)) return { ok: false, error: 'Type the amount paid' };
  var date = jobStr_(req.date) || consignToday_();
  if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) return { ok: false, error: 'Date must look like 2026-10-20' };
  var method = String(req.method || 'cash').toLowerCase();
  if ({ cash: 1, gcash: 1, bank: 1, other: 1 }[method] !== 1) method = 'other';
  var reference = consignClean_(req.reference, 80), note = consignClean_(req.note, 300);
  var lineId = jobStr_(req.line_id), cid = jobStr_(req.consignment_id), lines = [], blank = false;
  if (lineId) {
    var row = consignFind_('ConsignmentItems', ['line_id', 'id'], lineId);
    if (!row) return { ok: false, error: 'No such line' };
    if (consignNormStatus_(consignCell_(row, ['status'])) !== 'sold') return { ok: false, error: 'Record a payment on a sold piece' };
    if (consignOwedOf_(row) == null) return { ok: false, error: 'This sold piece has no agreed value, so there is no amount owed yet' };
    if (consignRemaining_(row) <= 0.009) return { ok: false, error: 'This piece is already paid' };
    lines = [row];
    cid = jobStr_(consignCell_(row, ['consignment_id']));
  } else if (cid) {
    jobRows_('ConsignmentItems').forEach(function (r) {
      if (jobStr_(consignCell_(r, ['consignment_id'])) !== cid) return;
      if (consignNormStatus_(consignCell_(r, ['status'])) !== 'sold') return;
      if (consignOwedOf_(r) == null) { blank = true; return; }
      if (consignRemaining_(r) > 0.009) lines.push(r);
    });
    lines.sort(function (a, b) {
      return consignCmp_(consignDay_(consignCell_(a, ['date_out'])), consignDay_(consignCell_(b, ['date_out'])))
        || consignCmp_(jobStr_(consignCell_(a, ['line_id', 'id'])), jobStr_(consignCell_(b, ['line_id', 'id'])));
    });
    if (!lines.length) return { ok: false, error: blank ? 'A sold piece has no agreed value, so there is no amount to record yet' : 'Nothing sold on this sheet still has a balance' };
  } else return { ok: false, error: 'Pick a sold piece or a sheet' };
  var room = 0;
  lines.forEach(function (r) { room = consignRound_(room + consignRemaining_(r)); });
  if (amount > room + 0.5) return { ok: false, error: 'That is more than the balance' };
  if (amount > room) amount = room;
  var left = amount, t = now_(), by = user.username, applied = [], payer = '';
  lines.forEach(function (r) {
    if (left <= 0.009) return;
    var rem = consignRemaining_(r);
    if (rem <= 0.009) return;
    var take = consignRound_(Math.min(left, rem));
    if (take <= 0) return;
    var prev = consignPaidOf_(r), next = consignRound_(prev + take);
    var lid = jobStr_(consignCell_(r, ['line_id', 'id']));
    var scid = jobStr_(consignCell_(r, ['consignment_id']));
    var who = consignPayer_(r);
    if (!payer) payer = who;
    var bal = Math.max(0, consignRound_((consignOwedOf_(r) || 0) - next));
    consignUpdate_('ConsignmentItems', r._row, { paid_amount: next, paid: next, paid_at: date, updated_at: t, updated_by: by });
    r.paid_amount = next;
    audit_(by, 'consignPay', lid, req.device_id, 'confirmed · payer ' + who + ' · ref ' + scid + ' ' + lid + ' · ' + take + ' ' + method + (reference ? ' · receipt ' + reference : '') + (note ? ' · ' + note : '') + ' · paid ' + next + ' · balance ' + bal);
    applied.push({ line_id: lid, consignment_id: scid, amount: take, paid: next, balance: bal });
    left = consignRound_(left - take);
  });
  if (!applied.length) return { ok: false, error: 'Nothing to apply' };
  consignSyncClose_(cid, user, t);
  return { ok: true, status: 'confirmed', payer: payer, applied: applied, reference: reference };
}

function consignSheets_(people, today, limit, exec) {
  var byId = {}; people.forEach(function (p) { byId[p.consignee_id] = p; });
  var items = {}, order = [];
  jobRows_('ConsignmentItems').forEach(function (r) {
    var cid = jobStr_(consignCell_(r, ['consignment_id'])); if (!cid) return;
    if (!items[cid]) { items[cid] = []; order.push(cid); }
    items[cid].push(r);
  });
  var out = [], seen = {};
  jobRows_('Consignments').forEach(function (sh) {
    var cid = jobStr_(consignCell_(sh, ['consignment_id', 'id'])); if (!cid || seen[cid]) return; seen[cid] = true;
    out.push(consignSheetView_(sh, items[cid] || [], byId, today, limit, exec));
  });
  order.forEach(function (cid) { if (!seen[cid]) out.push(consignSheetView_({ consignment_id: cid }, items[cid], byId, today, limit, exec)); });
  out.sort(function (a, b) { return consignCmp_(a.date_out, b.date_out) || consignCmp_(a.consignment_id, b.consignment_id); });
  return out;
}
function consignSheetView_(sh, rows, byId, today, limit, exec) {
  var gid = jobStr_(consignCell_(sh, ['consignee_id'])), person = byId[gid];
  var name = person ? person.name : jobStr_(consignCell_(sh, ['consignee_name', 'consignee', 'name']));
  var dateOut = consignDay_(consignCell_(sh, ['date_out', 'date'])) || consignDay_(rows[0] && consignCell_(rows[0], ['date_out']));
  var due = consignDay_(consignCell_(sh, ['due_date', 'due']));
  var roll = { sold: 0, owed: 0, paid: 0, balance: 0 };
  var viewItems = rows.map(function (r) {
    var st = consignNormStatus_(consignCell_(r, ['status']));
    var dout = consignDay_(consignCell_(r, ['date_out'])) || dateOut;
    var idue = consignDay_(consignCell_(r, ['due_date'])) || due;
    var open = st === 'consigned';
    var money = consignViewMoney_(r);
    var o = {
      line_id: jobStr_(consignCell_(r, ['line_id', 'id'])), item_id: jobStr_(consignCell_(r, ['item_id', 'sku', 'code'])),
      jo: jobKey_(consignCell_(r, ['jo'])) || jobStr_(consignCell_(r, ['jo'])), description: jobStr_(consignCell_(r, ['description', 'name'])),
      status: st, date_out: dout, due_date: idue, prev_location: consignLoc_(consignCell_(r, ['prev_location', 'previous_location'])),
      returned_at: jobStr_(consignCell_(r, ['returned_at'])), sold_at: jobStr_(consignCell_(r, ['sold_at'])),
      days_out: consignSpan_(dout, today), overdue: open && consignIsOverdue_(dout, idue, today, limit), _settled: money.settled
    };
    var val = consignCell_(r, ['value', 'agreed_value']);
    if (val !== '' && val != null) o.value = consignMoneySafe_(val);
    if (exec && st === 'sold') {
      o.owed = money.owed;
      o.paid = money.paid;
      o.balance = money.balance;
      if (money.paid_at) o.paid_at = money.paid_at;
      roll.sold++;
      if (money.owed != null) {
        roll.owed = consignRound_(roll.owed + money.owed);
        roll.balance = consignRound_(roll.balance + (money.balance || 0));
      }
      roll.paid = consignRound_(roll.paid + money.paid);
    }
    return o;
  });
  var settled = viewItems.length > 0 && viewItems.every(function (it) { return it._settled; });
  viewItems.forEach(function (it) { delete it._settled; });
  var anyOpen = viewItems.some(function (it) { return it.status === 'consigned'; });
  var stored = jobStr_(consignCell_(sh, ['status'])).toLowerCase();
  var view = {
    consignment_id: jobStr_(consignCell_(sh, ['consignment_id', 'id'])) || (rows[0] && jobStr_(consignCell_(rows[0], ['consignment_id']))),
    consignee_id: gid, consignee: name, phone: person ? person.phone : '', notes: jobStr_(consignCell_(sh, ['notes'])),
    date_out: dateOut, due_date: due, status: (!viewItems.length ? (stored === 'closed' ? 'closed' : 'open') : (settled ? 'closed' : 'open')),
    days_out: consignSpan_(dateOut, today), overdue: anyOpen && viewItems.some(function (it) { return it.overdue; }), items: viewItems
  };
  if (exec) { view.sold = roll.sold; view.owed = roll.owed; view.paid = roll.paid; view.balance = roll.balance; }
  return view;
}

    if (!consignItemOpen_(consignCell_(r, ['status']))) return;

function consignSpan_(from, to) {
  if (!from || !to) return null;

  if (isNaN(a) || isNaN(b)) return null;
  return Math.round((b - a) / 86400000);
}
