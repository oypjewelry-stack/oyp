PrintQueue.gs — a cheap print-queue revision so a station can poll every 5 seconds without reading the sheet.
CacheService key oypPrintRev. A brief printQueue returns {ok, brief:true, rev} and does not read PrintQueue, JobOrders, or Clients.
rev is null when the cache is empty or CacheService is missing. The station then keeps the 15 second full poll.
Bump the rev whenever a row is created, requeued, claimed, printed, failed, or cancelled.

var PRINT_REV_KEY = 'oypPrintRev';
function printCache_() {
  try {
    if (typeof CacheService === 'undefined' || !CacheService.getScriptCache) return null;
    return CacheService.getScriptCache();
  } catch (e) { return null; }
}
function printRevRead_() {
  var c = printCache_();
  if (!c) return null;
  var v = c.get(PRINT_REV_KEY);
  return v == null || v === '' ? null : String(v);
}
function printRevBump_() {
  var c = printCache_();
  if (!c) return;
  c.put(PRINT_REV_KEY, String(new Date().getTime()), 21600);
}
function printRevEnsure_() {
  var v = printRevRead_();
  if (v != null) return v;
  var c = printCache_();
  if (!c) return null;
  c.put(PRINT_REV_KEY, '1', 21600);
  return '1';
}
function printBrief_(req) {
  if (!req) return false;
  var b = req.brief;
  return b === 1 || b === true || b === '1' || b === 'true';
}
