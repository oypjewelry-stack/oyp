/**
 * Inventory bulk move / archive. One action: invBulk.
 * Sets JobOrders.location for each JO in the request. Writes updated_at and updated_by.
 * Archive is location "archived". Unarchive clears location. invBulk never deletes a row.
 * invDelete deletes one JobOrders row. The page asks first.
 *
 * Auth.gs handleAction_, immediately after the JobSync line:
 *   if (typeof isInvBulkAction_ === 'function' && isInvBulkAction_(a)) return handleInvBulk_(req, s.user); // InvBulk.gs
 */
var INV_BULK_ROLES = { admin: true, production_manager: true };

function isInvBulkAction_(a) { a = String(a || ''); return a === 'invBulk' || a === 'invDelete'; }

function handleInvBulk_(req, user) {
  if (!user || !user.username) {
    try {
      var sess = (typeof sessionFrom_ === 'function') ? sessionFrom_(req || {}) : null;
      if (sess && sess.user) user = sess.user;
      else if (sess && sess.username) user = sess;
    } catch (e) {}
  }
  if (!req || (req.action !== 'invBulk' && req.action !== 'invDelete')) return { ok: false, error: 'Unknown action' };
  if (!user || !INV_BULK_ROLES[user.role]) {
    try { audit_((user && user.username) || '', 'denied', 'invBulk', req.device_id, 'role ' + ((user && user.role) || '')); } catch (e2) {}
    return { ok: false, error: 'Not allowed', code: 'FORBIDDEN' };
  }
  var res;
  try { res = (typeof withLock_ === 'function') ? withLock_(function () { return req.action === 'invDelete' ? invDelete_(req, user) : invBulk_(req, user); }) : (req.action === 'invDelete' ? invDelete_(req, user) : invBulk_(req, user)); }
  catch (err) { res = { ok: false, error: String(err && err.message || err).slice(0, 300) }; }
  return res;
}

function invBulk_(req, user) {
  if (!ss_().getSheetByName('JobOrders')) return { ok: false, error: 'JobOrders tab is missing' };
  var loc = invBulkLoc_(req.location);
  if (loc.error) return loc;
  var jos = req.jos;
  if (!Array.isArray(jos) || !jos.length) return { ok: false, error: 'Select at least one item' };
  if (jos.length > 300) return { ok: false, error: 'Select at most 300 items at a time' };
  var allowed = invBulkAllowed_();
  if (loc.value && !allowed[loc.value]) return { ok: false, error: 'That location is not on the list' };
  var want = {}, order = [], i, raw;
  for (i = 0; i < jos.length; i++) {
    raw = jobStr_(jos[i]);
    if (!raw || want[raw]) continue;
    want[raw] = true;
    order.push(raw);
  }
  if (!order.length) return { ok: false, error: 'Select at least one item' };
  var byJo = {};
  jobRows_('JobOrders').forEach(function (j) {
    var id = jobStr_(j.jo);
    if (id && !byJo[id]) byJo[id] = j;
  });
  var t = (typeof now_ === 'function') ? now_() : new Date().toISOString();
  var by = user.username, set = [], missing = [], same = [];
  order.forEach(function (jo) {
    var row = byJo[jo];
    if (!row) { missing.push(jo); return; }
    var prev = jobStr_(row.location).toLowerCase();
    if (prev === loc.value) { same.push(jo); return; }
    var patch = { location: loc.value, updated_at: t, updated_by: by };
    invBulkStatus_(row, loc.value, patch);
    jobSet_('JobOrders', row._row, patch);
    try { audit_(by, loc.value === 'archived' ? 'invArchive' : (loc.value ? 'invMove' : 'invUnarchive'), jo, req.device_id, (prev || 'none') + ' -> ' + (loc.value || 'none')); } catch (e3) {}
    set.push({ jo: jo, location: loc.value, from: prev });
  });
  return { ok: true, location: loc.value, set: set, same: same, missing: missing };
}

function invBulkLoc_(v) {
  var s = jobStr_(v).toLowerCase();
  if (!s) return { value: '' };
  if (s === 'archived') return { value: 'archived' };
  if (!/^[a-z][a-z0-9_]{0,31}$/.test(s)) return { ok: false, error: 'That location is not on the list' };
  return { value: s };
}

function invBulkAllowed_() {
  var keys = { archived: 1 }, list = (typeof JOB_LOCATIONS !== 'undefined') ? JOB_LOCATIONS : null, i, item, k;
  if (typeof list === 'function') { try { list = list(); } catch (e) { list = null; } }
  if (list && typeof list === 'object') {
    if (typeof list.length === 'number') {
      for (i = 0; i < list.length; i++) {
        item = list[i];
        if (item == null) continue;
        if (typeof item === 'string') k = item;
        else if (typeof item.length === 'number') k = item[0];
        else k = item.key || item.id || item.location || '';
        k = jobStr_(k).toLowerCase();
        if (k) keys[k] = 1;
      }
    } else {
      Object.keys(list).forEach(function (name) {
        var child = list[name];
        k = jobStr_(child && (child.key || child.id) || name).toLowerCase();
        if (/^[a-z][a-z0-9_]{0,31}$/.test(k)) keys[k] = 1;
      });
    }
  }
  if (ss_().getSheetByName('JobOrders')) jobRows_('JobOrders').forEach(function (j) {
    k = jobStr_(j.location).toLowerCase();
    if (k) keys[k] = 1;
  });
  return keys;
}

function invDelete_(req, user) {
  if (!ss_().getSheetByName('JobOrders')) return { ok: false, error: 'JobOrders tab is missing' };
  var jo = jobStr_(req.jo);
  if (!jo) return { ok: false, error: 'Missing JO' };
  var row = null;
  jobRows_('JobOrders').forEach(function (j) {
    if (!row && jobStr_(j.jo) === jo) row = j;
  });
  if (!row || !row._row) return { ok: false, error: 'Not found' };
  ss_().getSheetByName('JobOrders').deleteRow(row._row);
  try { audit_(user.username, 'invDelete', jo, req.device_id, ''); } catch (e) {}
  return { ok: true, jo: jo };
}

function invBulkStatus_(row, loc, patch) {
  if (!loc || typeof JOB_LOC_STATUS === 'undefined' || !JOB_LOC_STATUS || !JOB_LOC_STATUS[loc]) return;
  var st = jobStr_(row.status);
  if (st && st !== 'draft') return;
  if (jobHeader_(ss_().getSheetByName('JobOrders')).indexOf('status') < 0) return;
  patch.status = JOB_LOC_STATUS[loc];
}
